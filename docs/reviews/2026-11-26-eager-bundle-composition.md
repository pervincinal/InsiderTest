Analysis only; no source, config or test was changed. Author: Frontend. Scripts used are in the appendix.

## 1. What "eager" means (scripts/checkBundle.mjs)

- `checkBundle` reads dist/index.html and sums gzip (level 9) of every `<script src>` and `<link rel="modulepreload">` ending in `.js`. Dynamic `import()` chunks are not in that list.
- Today index.html has 1 script and 0 modulepreload links, so eager = the entry chunk only. vite.config.ts has no `manualChunks`; the only chunk rule is the file name of the AI chunk (PERF-9).
- Consequence: a new chunk reached only through `import()` is **not counted** by `checkBundle`, however early it is warmed (lazyScreens, ai, result, hudOverlays are warmed at idle). A chunk reached through a static import from the entry would become a modulepreload and be counted. The budget therefore measures "bytes before first frame", not "bytes downloaded in the first seconds"; moving code to an idle-warmed chunk lowers the number without lowering total download.
- Lazy boundaries in use: `src/ui/lazyScreens.ts` (level map, shop, achievements, settings, how-to, title idle painter, focus ring, screen slide, save notices), `src/lazyChunk.ts` `loadChunk` wraps every dynamic import with retry; level bodies (src/levels/manifest.ts), AI (src/ui/ai.ts), result (src/ui/resultChunk.ts), tutorial steps, store, debug (`__DEBUG_SURFACE__`), perfOverlay, locales.

## 4. Approximations and caveats

- Marginal gzip assumes the module's removal also removes no further code; removing an export edge can tree-shake more (upside) and adds loader glue (downside, about 60-120 B per dynamic import).
- Source-map attribution is per mapping segment; code the minifier hoisted or merged may be attributed to a neighbouring module. Raw totals are exact; gzip numbers are estimates (+-5 %).
- Reachability is a static-import approximation with regex parsing; it ignores `import type` (correct) and treats a cut as removing the edge from every importer. It does not prove the module can leave without code changes: the "import change" lines above name what has to change.
- Nothing was built or measured with the changes applied.

### marginal.cjs
```js
// usage: node marginal.cjs <chunk.js> ; prints per source: raw, standalone gz, marginal gz (full gz - gz with that source's segments removed)
const fs=require('fs'),zlib=require('zlib');const {SourceMapConsumer}=require('source-map-js');
const f=process.argv[2];const code=fs.readFileSync(f,'utf8');const c=new SourceMapConsumer(JSON.parse(fs.readFileSync(f+'.map','utf8')));
const lines=code.split('\n');const st=[];let o=0;for(const l of lines){st.push(o);o+=l.length+1;}
const ms=[];c.eachMapping(m=>ms.push(m));ms.sort((a,b)=>a.generatedLine-b.generatedLine||a.generatedColumn-b.generatedColumn);
const ranges={};
for(let i=0;i<ms.length;i++){const a=st[ms[i].generatedLine-1]+ms[i].generatedColumn;const b=i+1<ms.length?st[ms[i+1].generatedLine-1]+ms[i+1].generatedColumn:code.length;(ranges[ms[i].source]??=[]).push([a,b]);}
const gz=s=>zlib.gzipSync(s,{level:9}).length;const full=gz(code);
const res=[];
for(const [src,rs] of Object.entries(ranges)){
  let out='',p=0,own='';rs.sort((x,y)=>x[0]-y[0]);
  for(const [a,b] of rs){out+=code.slice(p,a);own+=code.slice(a,b);p=b;}out+=code.slice(p);
  res.push({src:src.replace(/^(\.\.\/)+home\/user\/InsiderTest\/tower-clash\//,''),raw:Buffer.byteLength(own),alone:gz(own),marginal:full-gz(out)});
}
res.sort((a,b)=>b.marginal-a.marginal);
console.log(JSON.stringify({full,mods:res}));
```

### reach.cjs
```js
// usage: node reach.cjs <marg.json> <cut1,cut2,...>  (cut = src-relative file paths whose import edges FROM main.ts are dropped, e.g. src/ui/play.ts)
const fs=require('fs'),path=require('path');
const root=process.cwd();const marg=require(process.argv[2]);const m=Object.fromEntries(marg.mods.map(x=>[x.src,x]));
const cuts=new Set((process.argv[3]||'').split(',').filter(Boolean));
const res=(from,spec)=>{if(!spec.startsWith('.'))return null;const b=path.resolve(path.dirname(path.join(root,from)),spec);for(const c of [b+'.ts',path.join(b,'index.ts')])if(fs.existsSync(c))return path.relative(root,c);return null;};
const deps=f=>{const s=fs.readFileSync(path.join(root,f),'utf8');const o=[];for(const x of s.matchAll(/^\s*(?:import|export)\s+(?!type\b)(?:[^;'"]*?\s+from\s+)?'([^']+)'/gm)){const t=res(f,x[1]);if(t)o.push(t);}return o;};
const seen=new Set();const st=['src/main.ts'];
while(st.length){const f=st.pop();if(seen.has(f))continue;seen.add(f);for(const d of deps(f)){if(cuts.has(d))continue;st.push(d);}}
let kept=0,total=0,cutList=[];
```

### attrib.cjs (raw and gzip-alone per module)
```js
// usage: node attrib.cjs <chunk.js> ; needs source-map-js resolvable via NODE_PATH
const fs=require('fs'),zlib=require('zlib'),path=require('path');
const {SourceMapConsumer}=require('source-map-js');
const f=process.argv[2];const code=fs.readFileSync(f,'utf8');
const map=JSON.parse(fs.readFileSync(f+'.map','utf8'));
const c=new SourceMapConsumer(map);
const lines=code.split('\n');const starts=[];let o=0;for(const l of lines){starts.push(o);o+=l.length+1;}
const ms=[];c.eachMapping(m=>ms.push(m));
ms.sort((a,b)=>a.generatedLine-b.generatedLine||a.generatedColumn-b.generatedColumn);
const segs={};let attributed=0;
function add(src,s){(segs[src]??=[]).push(s);}
for(let i=0;i<ms.length;i++){
  const m=ms[i];const a=starts[m.generatedLine-1]+m.generatedColumn;
  const b=i+1<ms.length?starts[ms[i+1].generatedLine-1]+ms[i+1].generatedColumn:code.length;
  add(m.source||'(none)',code.slice(a,b));
}
const first=ms.length?starts[ms[0].generatedLine-1]+ms[0].generatedColumn:0;
if(first>0)add('(prelude)',code.slice(0,first));
const out=Object.entries(segs).map(([s,a])=>{const j=a.join('');return{src:s.replace(/^(\.\.\/)+/,''),raw:Buffer.byteLength(j),gz:zlib.gzipSync(j,{level:9}).length}});
out.sort((a,b)=>b.gz-a.gz);
console.log(JSON.stringify({total:{raw:Buffer.byteLength(code),gz:zlib.gzipSync(code,{level:9}).length,sumGz:out.reduce((x,y)=>x+y.gz,0),sumRaw:out.reduce((x,y)=>x+y.raw,0)},mods:out}));
```
